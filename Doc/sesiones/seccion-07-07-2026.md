**SQL ejecutado en Supabase:**
```sql
-- Columnas nuevas en productos
ALTER TABLE productos ADD COLUMN IF NOT EXISTS origen TEXT DEFAULT 'admin';
ALTER TABLE productos ADD COLUMN IF NOT EXISTS proveedor_nombre TEXT DEFAULT NULL;

-- Recrear VIEW (Postgres no permite cambiar columnas con CREATE OR REPLACE)
DROP VIEW IF EXISTS productos_con_estado;
CREATE VIEW productos_con_estado AS
  SELECT p.*, c.nombre AS categoria_nombre,
    CASE
      WHEN p.stock = 0 THEN 'Sin stock'
      WHEN p.stock <= 5 THEN 'Stock bajo'
      ELSE 'Activo'
    END AS estado
  FROM productos p
  LEFT JOIN categorias c ON p.categoria_id = c.id;

-- Fix datos existentes (ejemplo)
UPDATE productos SET origen = 'proveedor', proveedor_nombre = 'talkiesexpress'
WHERE nombre IN ('pinzas para jardineria', 'Ventilador de turbina');
```
